
# Rychlý tahák

```bash
# inicializace
cd repozitar
git init

# stav
git status

# přidání změn
git add -A

# vytvoření commitu
git commit -m "Popis"

# historie
git log --oneline

# větve
git branch
git checkout -b nova-vetev
git merge nova-vetev

# vzdálený repozitář
git remote -v

# synchronizace
git fetch
git pull
git push

# ssh test
ssh -T git@gitlab.panska.cz
```
